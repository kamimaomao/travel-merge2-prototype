# Hex Kingdoms Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a standalone mobile-first `hex-kingdoms` Slay-like web prototype with a complete human-versus-AI quick match and PNG-only board art.

**Architecture:** Create a new Vite + React + TypeScript app in `hex-kingdoms/`, leaving the existing Merge-2 prototype untouched. Keep all rules in pure TypeScript modules under `src/game/`, render the portrait UI with React components, and render board objects only with generated PNG assets through `<img>`.

**Tech Stack:** Vite, React 18, TypeScript, Vitest, Testing Library, CSS, image2/image_gen PNG assets.

```ts
import { BOARD_COLS, BOARD_ROWS } from "./rules";
import type { GameState, HexId, HexTile, Owner } from "./types";

const AXIAL_DIRECTIONS = [
  [1, 0],
  [1, -1],
  [0, -1],
  [-1, 0],
  [-1, 1],
  [0, 1]
] as const;

const SEA_IDS = new Set<HexId>([
  "0,0",
  "1,0",
  "5,0",
  "6,0",
  "0,1",
  "6,1",
  "0,2",
  "6,2",
  "0,3",
  "6,3",
  "0,4",
  "6,4",
  "0,5",
  "6,5",
  "0,6",
  "6,6",
  "0,7",
  "6,7",
  "0,8",
  "1,8",
  "5,8",
  "6,8"
]);

const PLAYER_IDS = new Set<HexId>(["1,2", "2,2", "1,3", "2,3", "1,4", "2,4"]);
const AI_IDS = new Set<HexId>(["4,4", "5,4", "4,5", "5,5", "4,6", "5,6"]);

export function hexId(q: number, r: number): HexId {
  return `${q},${r}` as HexId;
}

export function createBaseTiles(): Record<HexId, HexTile> {
  const tiles: Record<HexId, HexTile> = {};
  for (let r = 0; r < BOARD_ROWS; r += 1) {
    for (let q = 0; q < BOARD_COLS; q += 1) {
      const id = hexId(q, r);
      const terrain = SEA_IDS.has(id) ? "sea" : "land";
      let owner: Owner = "neutral";
      if (terrain === "sea") {
        owner = "neutral";
      } else if (PLAYER_IDS.has(id)) {
        owner = "player";
      } else if (AI_IDS.has(id)) {
        owner = "ai";
      }
      tiles[id] = { id, q, r, terrain, owner, piece: null, treasuryId: null };
    }
  }
  return tiles;
}

export function getNeighborIds(state: GameState, tileId: HexId): HexId[] {
  const tile = state.tiles[tileId];
  if (!tile) {
    return [];
  }
  return AXIAL_DIRECTIONS.map(([dq, dr]) => hexId(tile.q + dq, tile.r + dr)).filter((id) => Boolean(state.tiles[id]));
}

export function landTileIds(state: GameState): HexId[] {
  return Object.values(state.tiles)
    .filter((tile) => tile.terrain === "land")
    .map((tile) => tile.id);
}
```

```ts
import { INCOME_PER_OPEN_TILE, PIECE_RULES } from "./rules";
import { getNeighborIds } from "./map";
import type { BoardPiece, GameState, HexId, Side, Territory } from "./types";

function cloneState(state: GameState): GameState {
  return {
    ...state,
    tiles: Object.fromEntries(Object.entries(state.tiles).map(([id, tile]) => [id, { ...tile, piece: tile.piece ? { ...tile.piece } : null }])),
    treasuries: Object.fromEntries(Object.entries(state.treasuries).map(([id, treasury]) => [id, { ...treasury }]))
  };
}

function unitWage(piece: BoardPiece | null): number {
  return piece?.kind === "unit" ? PIECE_RULES.units[piece.level].wage : 0;
}

function tileProducesIncome(piece: BoardPiece | null): boolean {
  return piece?.kind !== "tree";
}

function connectedComponents(state: GameState, side: Side): HexId[][] {
  const remaining = new Set(
    Object.values(state.tiles)
      .filter((tile) => tile.terrain === "land" && tile.owner === side)
      .map((tile) => tile.id)
  );
  const components: HexId[][] = [];
  while (remaining.size > 0) {
    const first = remaining.values().next().value as HexId;
    const queue = [first];
    const component: HexId[] = [];
    remaining.delete(first);
    while (queue.length > 0) {
      const current = queue.shift() as HexId;
      component.push(current);
      for (const neighborId of getNeighborIds(state, current)) {
        const neighbor = state.tiles[neighborId];
        if (neighbor?.terrain === "land" && neighbor.owner === side && remaining.has(neighborId)) {
          remaining.delete(neighborId);
          queue.push(neighborId);
        }
      }
    }
    components.push(component);
  }
  return components;
}

export function normalizeTreasuries(state: GameState): GameState {
  const next = cloneState(state);
  for (const side of ["player", "ai"] as const) {
    for (const component of connectedComponents(next, side)) {
      const treasuryIds = component.map((tileId) => next.tiles[tileId].treasuryId).filter(Boolean) as string[];
      const primaryId = treasuryIds[0] ?? `${side}-treasury-${next.nextTreasuryNumber++}`;
      if (!next.treasuries[primaryId]) {
        next.treasuries[primaryId] = { id: primaryId, side, tileId: component[0], money: 0 };
      }
      for (const tileId of component) {
        next.tiles[tileId].treasuryId = primaryId;
      }
      next.treasuries[primaryId].tileId = component.includes(next.treasuries[primaryId].tileId) ? next.treasuries[primaryId].tileId : component[0];
      for (const duplicateId of treasuryIds.slice(1)) {
        next.treasuries[primaryId].money += next.treasuries[duplicateId]?.money ?? 0;
        delete next.treasuries[duplicateId];
      }
    }
  }
  return next;
}

export function getTerritoriesForSide(state: GameState, side: Side): Territory[] {
  const normalized = normalizeTreasuries(state);
  return connectedComponents(normalized, side).map((tileIds) => {
    const treasuryId = normalized.tiles[tileIds[0]].treasuryId as string;
    const income = tileIds.filter((tileId) => tileProducesIncome(normalized.tiles[tileId].piece)).length * INCOME_PER_OPEN_TILE;
    const wages = tileIds.reduce((sum, tileId) => sum + unitWage(normalized.tiles[tileId].piece), 0);
    return {
      id: treasuryId,
      side,
      tileIds,
      treasuryId,
      money: normalized.treasuries[treasuryId].money,
      income,
      wages
    };
  });
}
```

```ts
import { GRAVE_TURNS_BEFORE_TREE } from "./rules";
import { getNeighborIds } from "./map";
import { getTerritoriesForSide, normalizeTreasuries } from "./territory";
import type { GameState, HexId, Side } from "./types";

function cloneState(state: GameState): GameState {
  return {
    ...state,
    tiles: Object.fromEntries(Object.entries(state.tiles).map(([id, tile]) => [id, { ...tile, piece: tile.piece ? { ...tile.piece } : null }])),
    treasuries: Object.fromEntries(Object.entries(state.treasuries).map(([id, treasury]) => [id, { ...treasury }]))
  };

export function beginSideTurn(state: GameState, side: Side): GameState {
  const next = normalizeTreasuries(cloneState(state));
  for (const territory of getTerritoriesForSide(next, side)) {
    const treasury = next.treasuries[territory.treasuryId];
    const afterIncome = treasury.money + territory.income;
    if (afterIncome < territory.wages) {
      treasury.money = 0;
      for (const tileId of territory.tileIds) {
        const piece = next.tiles[tileId].piece;
        if (piece?.kind === "unit") {
          next.tiles[tileId].piece = { kind: "grave", age: 0 };
        }
      }
      next.message = `${side === "player" ? "Your" : "The rival"} territory went bankrupt.`;
    } else {
      treasury.money = afterIncome - territory.wages;
    }
    for (const tileId of territory.tileIds) {
      const piece = next.tiles[tileId].piece;
      if (piece?.kind === "unit" && piece.side === side) {
        piece.hasActed = false;
      }
    }
  }
  return next;
}

export function spreadTrees(state: GameState): GameState {
  const next = cloneState(state);
  const newTreeIds = new Set<HexId>();
  for (const tile of Object.values(state.tiles)) {
    if (tile.piece?.kind === "grave" && tile.piece.age >= GRAVE_TURNS_BEFORE_TREE) {
      newTreeIds.add(tile.id);
    } else if (tile.piece?.kind === "grave") {
      next.tiles[tile.id].piece = { kind: "grave", age: tile.piece.age + 1 };
    }
    if (tile.piece?.kind === "tree") {
      const target = getNeighborIds(state, tile.id).find((neighborId) => {
        const neighbor = state.tiles[neighborId];
        return neighbor.terrain === "land" && neighbor.piece === null;
      });
      if (target) {
        newTreeIds.add(target);
      }
    }
  }
  for (const tileId of newTreeIds) {
    next.tiles[tileId].piece = { kind: "tree" };
  }
  return next;
}
```

```ts
import { getNeighborIds } from "./map";
import { PIECE_RULES, purchaseCost } from "./rules";
import { getTerritoriesForSide, normalizeTreasuries } from "./territory";
import type { GameState, HexId, PurchaseKind, Side, UnitLevel } from "./types";

function cloneState(state: GameState): GameState {
  return {
    ...state,
    tiles: Object.fromEntries(Object.entries(state.tiles).map(([id, tile]) => [id, { ...tile, piece: tile.piece ? { ...tile.piece } : null }])),
    treasuries: Object.fromEntries(Object.entries(state.treasuries).map(([id, treasury]) => [id, { ...treasury }]))
  };
}

function sideLabel(side: Side): string {
  return side === "player" ? "Player" : "Rival";
}

function targetDefense(state: GameState, targetId: HexId): number {
  const target = state.tiles[targetId];
  let defense = target.treasuryId ? PIECE_RULES.treasury.defense : 0;
  if (target.piece?.kind === "unit") {
    defense = Math.max(defense, PIECE_RULES.units[target.piece.level].defense);
  }
  if (target.piece?.kind === "fort") {
    defense = Math.max(defense, PIECE_RULES.fort.defense);
  }
  for (const neighborId of getNeighborIds(state, targetId)) {
    const neighbor = state.tiles[neighborId];
    if (neighbor.owner === target.owner && neighbor.piece?.kind === "fort") {
      defense = Math.max(defense, PIECE_RULES.fort.defense);
    }
  }
  return defense;
}

function territoryForTile(state: GameState, side: Side, tileId: HexId) {
  return getTerritoriesForSide(state, side).find((territory) => territory.tileIds.includes(tileId));
}

export function buyPiece(state: GameState, side: Side, kind: PurchaseKind, tileId: HexId): GameState {
  const next = normalizeTreasuries(cloneState(state));
  const tile = next.tiles[tileId];
  const territory = territoryForTile(next, side, tileId);
  const cost = purchaseCost(kind);
  if (!tile || tile.terrain !== "land" || tile.owner !== side || tile.piece || !territory) {
    return { ...next, message: "Choose an empty tile in that territory." };
  }
  const treasury = next.treasuries[territory.treasuryId];
  if (treasury.money < cost) {
    return { ...next, message: "That territory does not have enough money." };
  }
  treasury.money -= cost;
  tile.piece =
    kind === "fort"
      ? { kind: "fort", side }
      : { kind: "unit", side, level: Number(kind.slice("unit-".length)) as UnitLevel, hasActed: false };
  return { ...next, pendingPurchase: null, selectedTileId: tileId, message: `${sideLabel(side)} bought ${kind}.` };
}

export function combineUnits(state: GameState, fromId: HexId, toId: HexId): GameState {
  const next = normalizeTreasuries(cloneState(state));
  const from = next.tiles[fromId];
  const to = next.tiles[toId];
  if (from.piece?.kind !== "unit" || to.piece?.kind !== "unit" || from.piece.side !== to.piece.side || from.piece.level !== to.piece.level || from.piece.level === 4) {
    return { ...next, message: "Those pieces cannot combine." };
  }
  const side = from.piece.side;
  if (!territoryForTile(next, side, fromId)?.tileIds.includes(toId)) {
    return { ...next, message: "Pieces must be in the same territory." };
  }
  to.piece = { kind: "unit", side, level: (from.piece.level + 1) as UnitLevel, hasActed: true };
  from.piece = null;
  return { ...next, selectedUnitId: toId, selectedTileId: toId, message: "Pieces combined." };
}

export function performUnitAction(state: GameState, fromId: HexId, targetId: HexId): GameState {
  const next = normalizeTreasuries(cloneState(state));
  const from = next.tiles[fromId];
  const target = next.tiles[targetId];
  if (from.piece?.kind !== "unit" || from.piece.hasActed || !target || target.terrain !== "land") {
    return { ...next, message: "That unit cannot act." };
  }
  if (!getNeighborIds(next, fromId).includes(targetId)) {
    return { ...next, message: "Choose an adjacent tile." };
  }
  const unit = from.piece;
  if (target.owner === unit.side && target.piece?.kind === "tree") {
    target.piece = null;
    unit.hasActed = true;
    return { ...next, message: "Tree chopped." };
  }
  if (target.owner === unit.side && target.piece === null) {
    target.piece = { ...unit, hasActed: true };
    from.piece = null;
    return { ...next, selectedUnitId: targetId, selectedTileId: targetId, message: "Unit moved." };
  }
  const attack = PIECE_RULES.units[unit.level].attack;
  if (attack <= targetDefense(next, targetId)) {
    return { ...next, message: "That target is too strong." };
  }
  target.owner = unit.side;
  target.treasuryId = from.treasuryId;
  target.piece = { ...unit, hasActed: true };
  from.piece = null;
  return normalizeTreasuries({ ...next, selectedUnitId: targetId, selectedTileId: targetId, message: "Land captured." });
}

export function checkWinner(state: GameState): Side | null {
  const hasPlayerLand = Object.values(state.tiles).some((tile) => tile.terrain === "land" && tile.owner === "player");
  const hasAiLand = Object.values(state.tiles).some((tile) => tile.terrain === "land" && tile.owner === "ai");
  if (hasPlayerLand && !hasAiLand) {
    return "player";
  }
  if (hasAiLand && !hasPlayerLand) {
    return "ai";
  }
  return null;
}
```

```ts
import { performUnitAction, buyPiece, checkWinner } from "./actions";
import { beginSideTurn, spreadTrees } from "./economy";
import { getNeighborIds } from "./map";
import { getTerritoriesForSide } from "./territory";
import type { GameState, HexId, Side } from "./types";

function enemyOf(side: Side): Side {
  return side === "player" ? "ai" : "player";
}

export function advanceTurn(state: GameState): GameState {
  const nextSide = enemyOf(state.currentSide);
  const afterTrees = nextSide === "player" ? spreadTrees(state) : state;
  const afterEconomy = beginSideTurn({ ...afterTrees, currentSide: nextSide }, nextSide);
  return { ...afterEconomy, winner: checkWinner(afterEconomy) };
}

function legalAttackTargets(state: GameState, fromId: HexId): HexId[] {
  const from = state.tiles[fromId];
  if (from.piece?.kind !== "unit" || from.piece.side !== "ai" || from.piece.hasActed) {
    return [];
  }
  return getNeighborIds(state, fromId).filter((targetId) => {
    const target = state.tiles[targetId];
    return target.terrain === "land" && target.owner !== "ai";
  });
}

export function runAiTurn(state: GameState): GameState {
  let next = state.currentSide === "ai" ? state : { ...state, currentSide: "ai" as const };
  const log: string[] = [...next.actionLog];
  for (let step = 0; step < 6; step += 1) {
    const unitId = Object.values(next.tiles).find((tile) => legalAttackTargets(next, tile.id).length > 0)?.id;
    if (unitId) {
      const targetId = legalAttackTargets(next, unitId)[0];
      next = performUnitAction(next, unitId, targetId);
      log.push(`Rival captured or attacked ${targetId}.`);
      continue;
    }
    const territory = getTerritoriesForSide(next, "ai").find((item) => item.money >= 10);
    const buyTarget = territory?.tileIds.find((tileId) => next.tiles[tileId].piece === null);
    if (buyTarget) {
      next = buyPiece(next, "ai", "unit-1", buyTarget);
      log.push(`Rival bought a scout at ${buyTarget}.`);
    }
    break;
  }
  const winner = checkWinner(next);
  if (winner) {
    return { ...next, winner, actionLog: log, message: winner === "player" ? "You win." : "The rival wins." };
  }
  const playerTurn = advanceTurn({ ...next, currentSide: "ai", actionLog: log });
  return { ...playerTurn, currentSide: "player", turn: state.turn + 1, actionLog: log, message: "Your turn." };
}
```

- [ ] **Step 5: Wire App state and actions**

Replace `hex-kingdoms/src/App.tsx`:

```tsx
import { useMemo, useState } from "react";
import { ActionPanel } from "./components/ActionPanel";
import { HexBoard } from "./components/HexBoard";
import { buyPiece, combineUnits, performUnitAction } from "./game/actions";
import { runAiTurn, advanceTurn } from "./game/ai";
import { createInitialState } from "./game/createInitialState";
import { getTerritoriesForSide } from "./game/territory";
import type { GameState, HexId, PurchaseKind } from "./game/types";

function selectedTerritoryFor(state: GameState) {
  if (!state.selectedTileId) {
    return null;
  }
  return getTerritoriesForSide(state, "player").find((territory) => territory.tileIds.includes(state.selectedTileId as HexId)) ?? null;
}

export default function App() {
  const [game, setGame] = useState<GameState>(() => createInitialState());
  const selectedTerritory = useMemo(() => selectedTerritoryFor(game), [game]);

  function handleTileClick(tileId: HexId) {
    setGame((current) => {
      if (current.winner) {
        return current;
      }
      const tile = current.tiles[tileId];
      if (current.pendingPurchase) {
        return buyPiece(current, "player", current.pendingPurchase, tileId);
      }
      if (current.selectedUnitId && current.selectedUnitId !== tileId) {
        const selectedPiece = current.tiles[current.selectedUnitId].piece;
        if (selectedPiece?.kind === "unit" && tile.piece?.kind === "unit" && tile.piece.side === selectedPiece.side && tile.piece.level === selectedPiece.level) {
          return combineUnits(current, current.selectedUnitId, tileId);
        }
        return performUnitAction(current, current.selectedUnitId, tileId);
      }
      if (tile.piece?.kind === "unit" && tile.piece.side === "player") {
        return { ...current, selectedTileId: tileId, selectedUnitId: tileId, message: "Choose an adjacent tile." };
      }
      return { ...current, selectedTileId: tileId, selectedUnitId: null, message: "Tile selected." };
    });
  }

  function handleBuy(kind: PurchaseKind) {
    setGame((current) => ({ ...current, pendingPurchase: kind, message: "Choose a friendly empty tile." }));
  }

  function handleEndTurn() {
    setGame((current) => runAiTurn(advanceTurn(current)));
  }

  return (
    <main className="app-shell">
      <section className="game-frame" aria-label="Hex Kingdoms prototype">
        <header className="top-hud">
          <div>
            <p className="eyebrow">Hex Kingdoms</p>
            <h1>Turn {game.turn}</h1>
          </div>
          <div className="hud-stat">
            <span>Side</span>
            <strong>{game.currentSide === "player" ? "You" : "Rival"}</strong>
          </div>
        </header>

        <HexBoard state={game} onTileClick={handleTileClick} />

        <p className="message-strip">{game.winner ? (game.winner === "player" ? "You win." : "The rival wins.") : game.message}</p>

        <ActionPanel
          state={game}
          selectedTerritory={selectedTerritory}
          onBuy={handleBuy}
          onEndTurn={handleEndTurn}
          onClearSelection={() => setGame((current) => ({ ...current, selectedTileId: null, selectedUnitId: null, pendingPurchase: null }))}
        />
      </section>
    </main>
  );
}
```
